// ① 실제 시트 이름으로 변경 (구글 폼 연결 시 자동 생성됨)
const SHEET_NAME = '폼 응답 1';

// ② 장소별 핀ID 매핑 (FIXED_LOCATIONS의 id와 동일하게)
const PIN_ID_MAP = {
    '교무실':       'f1-office',
    '보건실':       'f1-nurse',
    '국기계양대 앞': 'f1-canteen',
    '창고':         'f1-storage',
    '과학실 앞':    'f2-science',
    '미술실':       'f2-art',
    '2층 복도':     'f2-cor',
    '컴퓨터실':     'f3-comp',
    '책방 벼리':    'f3-lib',
    '3층 복도':     'f3-cor',
    '회의실':       'f4-conf',
    '교장실':       'f4-prin',
    '옥상 테라스':  'f4-terrace',
};

// ③ 측정 항목 한→영 변환
const METRIC_MAP = {
    '조도': 'lux',
    '소음': 'db',
    '온도': 'temp',
    '습도': 'humi',
    '미세먼지': 'dust',
};

/**
 * GET 요청 처리 — HTML에서 fetchData() 호출 시 응답
 */
function doGet(e) {
    try {
        const ss    = SpreadsheetApp.getActiveSpreadsheet();
        const sheet = ss.getSheetByName(SHEET_NAME);

        if (!sheet) {
            return jsonResponse({ error: '시트를 찾을 수 없습니다: ' + SHEET_NAME });
        }

        const rows   = sheet.getDataRange().getValues();
        if (rows.length < 2) {
            return jsonResponse({ records: [] });
        }

        const header  = rows[0];  // 헤더 행
        const dataRows = rows.slice(1);

        // 헤더 인덱스 찾기
        const idx = {
            time:   findColIdx(header, ['타임스탬프', 'Timestamp']),
            floor:  findColIdx(header, ['층 선택', '층']),
            loc1:   findColIdx(header, ['장소 선택 (1층)', '장소 선택', '1층 장소']),
            loc2:   findColIdx(header, ['장소 선택 (2층)', '2층 장소']),
            loc3:   findColIdx(header, ['장소 선택 (3층)', '3층 장소']),
            loc4:   findColIdx(header, ['장소 선택 (4층)', '4층 장소']),
            metric: findColIdx(header, ['측정 항목', '측정항목']),
            value:  findColIdx(header, ['값 입력', '측정값', '값']),
            user:   findColIdx(header, ['측정자 이름', '측정자', '이름']),
            memo:   findColIdx(header, ['메모', '비고']),
        };

        const records = dataRows
            .filter(row => row[idx.time] && row[idx.metric] && row[idx.value] !== '')
            .map(row => {
                // 층 번호 추출 (예: "1층" → "1", "2" → "2")
                const floorRaw = String(row[idx.floor] || '').trim();
                const floor    = floorRaw.replace(/층$/, '').trim();

                // 장소: 층에 따라 해당 컬럼에서 가져오기
                let loc = '';
                if      (floor === '1' && idx.loc1 >= 0) loc = String(row[idx.loc1] || '').trim();
                else if (floor === '2' && idx.loc2 >= 0) loc = String(row[idx.loc2] || '').trim();
                else if (floor === '3' && idx.loc3 >= 0) loc = String(row[idx.loc3] || '').trim();
                else if (floor === '4' && idx.loc4 >= 0) loc = String(row[idx.loc4] || '').trim();

                // 장소를 못 찾은 경우 모든 장소 컬럼에서 비어있지 않은 값 탐색
                if (!loc) {
                    [idx.loc1, idx.loc2, idx.loc3, idx.loc4].forEach(ci => {
                        if (ci >= 0 && !loc) {
                            const v = String(row[ci] || '').trim();
                            if (v) loc = v;
                        }
                    });
                }

                // 핀 ID 매핑
                const pinId = PIN_ID_MAP[loc] || '';

                // 측정 항목 한→영 변환
                const metricRaw = String(row[idx.metric] || '').trim();
                const metric    = METRIC_MAP[metricRaw] || metricRaw.toLowerCase();

                // 타임스탬프 포맷
                const timeVal = row[idx.time];
                let time = '';
                if (timeVal instanceof Date) {
                    time = Utilities.formatDate(timeVal, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm');
                } else {
                    time = String(timeVal).trim();
                }

                return {
                    time,
                    floor,
                    loc,
                    pinId,
                    metric,
                    value: String(row[idx.value] || '').trim(),
                    user:  idx.user  >= 0 ? String(row[idx.user]  || '').trim() : '',
                    memo:  idx.memo  >= 0 ? String(row[idx.memo]  || '').trim() : '',
                };
            })
            .filter(r => r.loc && r.metric && r.value);  // 유효한 기록만

        return jsonResponse({ records });

    } catch (err) {
        return jsonResponse({ error: err.message, records: [] });
    }
}

/**
 * 헤더에서 컬럼 인덱스 찾기 (여러 후보 이름 중 일치하는 첫 번째)
 */
function findColIdx(header, candidates) {
    for (let i = 0; i < header.length; i++) {
        const h = String(header[i]).trim();
        if (candidates.some(c => h.includes(c) || c.includes(h))) return i;
    }
    return -1;
}

/**
 * JSON 응답 생성
 */
function jsonResponse(data) {
    return ContentService
        .createTextOutput(JSON.stringify(data))
        .setMimeType(ContentService.MimeType.JSON);
}

/**
 * ================================================================
 * [테스트용] 스크립트 에디터에서 직접 실행해 동작 확인
 * ================================================================
 */
function testDoGet() {
    const result = doGet({});
    const data   = JSON.parse(result.getContent());
    Logger.log('총 기록 수: ' + data.records.length);
    if (data.records.length > 0) {
        Logger.log('첫 번째 기록: ' + JSON.stringify(data.records[0]));
    }
    if (data.error) {
        Logger.log('오류: ' + data.error);
    }
}
